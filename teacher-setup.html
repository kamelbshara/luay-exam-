// =============================================
// Google Apps Script — نتائج اختبارات الفيزياء
// مدرسة خالد بن الوليد | الأستاذ: لؤي بني عطا
// =============================================

const SECRET_TOKEN = 'physics2026'; // يجب أن يطابق token في الاختبار

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    if (data.token !== SECRET_TOKEN) {
      return respond({ error: 'unauthorized' });
    }
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // -------- تسجيل بدء الاختبار --------
    if (data.action === 'start') {
      const sh = getOrCreate(ss, 'سجل_البدء',
        ['الوقت', 'الاسم', 'الرقم', 'الشعبة', 'رمز_المعلم', 'المتصفح']);
      sh.appendRow([new Date(), data.student.name, data.student.id,
                    data.student.section, data.teacherCode || '', data.ua || '']);
      return respond({ status: 'start_ok' });
    }

    // -------- حفظ نتيجة التسليم --------
    const pdfUrl = savePdf(data.pdfBase64, data.pdfFilename);

    // ورقة الملخص
    const sumSh = getOrCreate(ss, 'نتائج_ملخص',
      ['وقت_التسليم','الاسم','الرقم','الشعبة','الاختبار','النموذج',
       'الدرجة','من','النسبة%','الوقت(ث)','المخالفات','انتهى_الوقت','رابط_PDF']);
    sumSh.appendRow([
      data.submittedAt, data.student.name, data.student.id, data.student.section,
      data.exam, data.model || '', data.score, data.total,
      (data.percent || 0) + '%', data.durationSeconds || '',
      data.violations || 0, data.timedOut ? 'نعم' : 'لا', pdfUrl
    ]);

    // ورقة تفاصيل الإجابات
    const detSh = getOrCreate(ss, 'تفاصيل_الإجابات',
      ['الاسم','الرقم','الشعبة','الاختبار',
       'رقم_السؤال','نوع_السؤال','إجابة_الطالب','الإجابة_الصحيحة','صح_خطأ','الدرجة_المكتسبة']);
    (data.answers || []).forEach(a => {
      detSh.appendRow([
        data.student.name, data.student.id, data.student.section, data.exam,
        a.no, a.type || 'mcq', a.chosen || '', a.correct || '',
        a.isCorrect ? 'صح ✓' : 'خطأ ✗', a.earned != null ? a.earned : ''
      ]);
    });

    // ورقة المهارات
    const skillSh = getOrCreate(ss, 'المهارات',
      ['الاسم','الرقم','الشعبة','الاختبار','المهارة','الدرجة_المكتسبة','الدرجة_الكلية','النسبة%']);
    (data.skills || []).forEach(sk => {
      skillSh.appendRow([
        data.student.name, data.student.id, data.student.section, data.exam,
        sk.name, sk.earned, sk.total, sk.percent + '%'
      ]);
    });

    return respond({ status: 'ok', pdfSaved: !!pdfUrl });
  } catch (err) {
    return respond({ error: err.message });
  }
}

function doGet(e) {
  const q = e && e.parameter ? e.parameter : {};
  if (q.token !== SECRET_TOKEN) return respond({ error: 'unauthorized' });
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName('نتائج_ملخص');
  if (!sh || sh.getLastRow() < 2) return respond({ results: [] });
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, sh.getLastColumn()).getValues();
  const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  const results = rows.map(r => {
    const obj = {};
    headers.forEach((h, i) => { obj[h] = r[i]; });
    return obj;
  });
  return respond({ results });
}

// -------- مساعدات --------
function getOrCreate(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow(headers);
    sh.getRange(1, 1, 1, headers.length).setBackground('#4f46e5')
      .setFontColor('#ffffff').setFontWeight('bold');
  }
  return sh;
}

function savePdf(b64, filename) {
  try {
    if (!b64 || !filename) return '';
    let folder;
    const it = DriveApp.getFoldersByName('نتائج_اختبارات_فيزياء');
    folder = it.hasNext() ? it.next() : DriveApp.createFolder('نتائج_اختبارات_فيزياء');
    const bytes = Utilities.base64Decode(b64);
    const blob  = Utilities.newBlob(bytes, 'application/pdf', filename);
    const file  = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getUrl();
  } catch (e) { return ''; }
}

function respond(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
